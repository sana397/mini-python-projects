4. JavaScript Pop-ups and Functions

// Display Date
document.getElementById('datebox').value = new Date().toLocaleString();
// Factorial
var n = prompt('Enter n:');
var f = 1; for (var i=1;i<=n;i++) f*=i; alert(f);
// Multiplication Table
var n = prompt('Enter n:');
var out = '';
for (var i=1;i<=10;i++) out += n+' x '+i+' = '+(n*i)+'\n';
alert(out);
// Sum Numbers
var n = prompt('How many?');
var sum = 0;
for (var i=0;i<n;i++) sum += parseInt(prompt('Enter number')); alert(sum);
